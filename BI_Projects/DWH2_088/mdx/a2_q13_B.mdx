  NON EMPTY Crossjoin(
    {
      [dim_timemonth].[Time].[month_name].&[Jan],
      [dim_timemonth].[Time].[month_name].&[Feb],
      [dim_timemonth].[Time].[month_name].&[Mar]
    },
    {
      [Measures].[Exceed Days (any)]
    }
  ) ON COLUMNS,
  NON EMPTY [dim_city].[Geo].[city_name].Members ON ROWS
  FROM (
    SELECT
    {
      [dim_alertpeak].[Alert].[ALL].[AllMember].[Yellow],
      [dim_alertpeak].[Alert].[ALL].[AllMember].[None]
    } ON COLUMNS
    FROM [AirQ Cube]
  )
