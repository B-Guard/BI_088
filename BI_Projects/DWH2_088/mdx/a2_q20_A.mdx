SELECT
  NON EMPTY Crossjoin(
    [dim_param].[Param].[param_name],
    [Measures].[Avg Recorded Value]
  ) ON COLUMNS,
  NON EMPTY [dim_city].[Geo].[country_name] ON ROWS
  FROM (
    SELECT
    [dim_param].[Param].[ALL].[AllMember].[Scientific Study] ON COLUMNS
    FROM (
      SELECT
      [dim_city].[Geo].[ALL].[AllMember].[Western Europe] ON COLUMNS
      FROM [AirQ Cube]
    )
    WHERE [dim_timemonth].[Time].[ALL].[AllMember].[2024]
  )
  WHERE [dim_timemonth].[Time].[ALL].[AllMember].[2024]
