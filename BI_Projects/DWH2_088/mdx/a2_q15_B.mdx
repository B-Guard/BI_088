SELECT
  NON EMPTY Crossjoin(
    {
      [dim_timemonth].[Time].[year_num].&[2023],
      [dim_timemonth].[Time].[year_num].&[2024]
    },
    {
      [Measures].[Exceed Days (any)]
    }
  ) ON COLUMNS,
  NON EMPTY [dim_city].[Geo].[country_name] ON ROWS
  FROM (
    SELECT
    [dim_city].[Geo].[ALL].[AllMember].[Eastern Europe] ON COLUMNS
    FROM [AirQ Cube]
  )
